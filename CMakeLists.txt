cmake_minimum_required(VERSION 3.21)
project(GameVision VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "GameVision currently requires Windows 10/11 and Direct3D 11")
endif()
add_executable(GameVision WIN32 src/main.cpp)
target_compile_features(GameVision PRIVATE cxx_std_17)
target_compile_definitions(GameVision PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(GameVision PRIVATE d3d11 dxgi d3dcompiler user32 gdi32 shell32)
if(MSVC)
    target_compile_options(GameVision PRIVATE /W4 /permissive- /EHsc)
endif()
